│  🚀 COMMANDES ESSENTIELLES                                           │
└──────────────────────────────────────────────────────────────────────┘

  # Vérifier la configuration
  ./check.sh

  # Démarrer tout automatiquement
  ./start.sh

  # OU manuellement :

  # Terminal 1 - Backend
  cd backend
  python3 -m venv venv
  source venv/bin/activate
  pip install -r requirements.txt
  python main.py

  # Terminal 2 - Mobile
  cd mobile
  npm install
  npx expo start
